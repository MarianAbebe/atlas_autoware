cmake_minimum_required(VERSION 3.14)
project(atlas_map_integration_demo)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(autoware_lanelet2_utils REQUIRED)
find_package(autoware_lanelet2_extension REQUIRED)
find_package(autoware_map_msgs REQUIRED)
find_package(autoware_planning_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(lanelet2_core REQUIRED)
find_package(lanelet2_routing REQUIRED)
find_package(lanelet2_traffic_rules REQUIRED)
add_executable(map_routing_probe src/map_routing_probe.cpp)
target_compile_features(map_routing_probe PUBLIC cxx_std_17)
ament_target_dependencies(map_routing_probe rclcpp autoware_lanelet2_utils
  autoware_lanelet2_extension autoware_map_msgs autoware_planning_msgs geometry_msgs
  nav_msgs sensor_msgs visualization_msgs lanelet2_core lanelet2_routing lanelet2_traffic_rules)
install(TARGETS map_routing_probe DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/run_validation.py scripts/analyze_alignment.py DESTINATION lib/${PROJECT_NAME})
ament_package()
