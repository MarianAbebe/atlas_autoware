cmake_minimum_required(VERSION 3.14)
project(atlas_localization_closed_loop_demo)
find_package(ament_cmake REQUIRED)
set(deps rclcpp geometry_msgs nav_msgs sensor_msgs std_msgs std_srvs autoware_map_msgs
  autoware_control_msgs autoware_planning_msgs autoware_vehicle_msgs autoware_adapi_v1_msgs
  pcl_conversions tf2_ros)
foreach(dep IN LISTS deps)
  find_package(${dep} REQUIRED)
endforeach()
find_package(PCL REQUIRED COMPONENTS common filters)
set(ATLAS_AUTOWARE_ROOT "$ENV{ATLAS_AUTOWARE_ROOT}" CACHE PATH "Pinned Autoware source root")
if(NOT ATLAS_AUTOWARE_ROOT)
  set(ATLAS_AUTOWARE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../references/autoware")
endif()
set(ATLAS_NDT_SOURCE "${ATLAS_AUTOWARE_ROOT}/src/core/autoware_core/localization/autoware_ndt_scan_matcher" CACHE PATH "Pinned NDT fixtures")
add_executable(localization_fixture src/localization_fixture.cpp)
target_compile_features(localization_fixture PRIVATE cxx_std_17)
target_compile_options(localization_fixture PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(localization_fixture SYSTEM PRIVATE ${PCL_INCLUDE_DIRS} "${ATLAS_NDT_SOURCE}/test")
target_link_libraries(localization_fixture ${PCL_LIBRARIES})
ament_target_dependencies(localization_fixture ${deps})
install(TARGETS localization_fixture DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/localization_watchdog.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME})
ament_package()
